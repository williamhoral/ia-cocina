async function preguntarAbuelita() { const ingredientes = document.getElementById('ingredientesInput').value; const resultadoDiv = document.getElementById('resultado'); if (!ingredientes) { alert('Por favor, escribe algún ingrediente.'); return; } resultadoDiv.innerHTML = 'Pensando en qué cocinar... 👵✨'; try { const response = await fetch('/api/cooking', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ ingredientes: ingredientes }) // ¡Ya no enviamos la API KEY desde el diseño! }); const data = await response.json(); if (data.receta) { resultadoDiv.innerHTML = data.receta; } else { resultadoDiv.innerHTML = 'Error: ' + data.error; } } catch (error) { resultadoDiv.innerHTML = 'No se pudo conectar con la abuelita.'; } }
